222 void stop() {
223 bool expected = true;
224 if (!running_.compare_exchange_strong(expected, false)) return;
225
226 // Unblock accept by connecting once.
227 try {
228 static socket_runtime rt;
229 socket_t s = ::socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
230 if (s != invalid_socket_v) {
231 sockaddr_in addr{};
232 addr.sin_family = AF_INET;
233 addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
234 addr.sin_port = htons(port_);
235 ::connect(s, reinterpret_cast<sockaddr*>(&addr), sizeof(addr));
236 close_socket(s);
237 }
238 } catch (...) {
239 }
240
241 if (worker_.joinable()) worker_.join();
242 close_socket(listen_sock_);
243 listen_sock_ = invalid_socket_v;
244 }

246private:
248 void run() {
249 while (running_.load()) {
250 sockaddr_in peer{};
251#if defined(_WIN32)
252 int peer_len = sizeof(peer);
253#else
254 socklen_t peer_len = sizeof(peer);
255#endif
256 socket_t client = ::accept(listen_sock_, reinterpret_cast<sockaddr*>(&peer), &peer_len);
257 if (client == invalid_socket_v) {
258 if (!running_.load()) break;
259 continue;
260 }
261
262 std::string req;
263 req.reserve(4096);
264 char buf[4096];
265
266 // Read headers first.
267 while (req.find("\r\n\r\n") == std::string::npos) {
268#if defined(_WIN32)
269 int n = ::recv(client, buf, static_cast<int>(sizeof(buf)), 0);
270#else
271 int n = static_cast<int>(::recv(client, buf, sizeof(buf), 0));
272#endif
273 if (n <= 0) break;
274 req.append(buf, buf + n);
275 if (req.size() > 1024 * 1024) break;
276 }
277
278 // If Content-Length exists, read full body.
279 std::size_t header_end = req.find("\r\n\r\n");
280 std::size_t body_start = header_end == std::string::npos ? req.size() : header_end + 4;
281
282 std::size_t content_length = 0;
283 {
284 std::string headers = header_end == std::string::npos ? req : req.substr(0, header_end);
285 std::string lower = to_lower_copy(headers);
286 std::string key = "content-length:";
287 std::size_t pos = lower.find(key);
288 if (pos != std::string::npos) {
289 std::size_t line_end = lower.find("\r\n", pos);
290 std::size_t vpos = pos + key.size();
291 while (vpos < lower.size() && (lower[vpos] == ' ' || lower[vpos] == '\t')) ++vpos;
292 std::string_view v(lower.c_str() + vpos,
293 (line_end == std::string::npos ? lower.size() : line_end) - vpos);
294 try {
295 content_length = static_cast<std::size_t>(std::stoul(std::string(v)));
296 } catch (...) {
297 content_length = 0;
298 }
299 }
300 }
301
302 while (req.size() < body_start + content_length) {
303#if defined(_WIN32)
304 int n = ::recv(client, buf, static_cast<int>(sizeof(buf)), 0);
305#else
306 int n = static_cast<int>(::recv(client, buf, sizeof(buf), 0));
307#endif
308 if (n <= 0) break;
309 req.append(buf, buf + n);
310 if (req.size() > 1024 * 1024) break;
311 }
312
313 std::string resp;
314 {
315 std::lock_guard<std::mutex> lock(req_mu_);
316 last_request_ = req;
317 ++request_count_;
318 }
319 req_cv_.notify_all();
320 try {
321 resp = handler_ ? handler_(req) : build_http_response("500 Internal Server Error", "text/plain", "no handler");
322 } catch (const std::exception& e) {
323 resp = build_http_response("500 Internal Server Error", "text/plain", e.what());
324 }
325
326 // Best-effort send all.
327 std::size_t sent = 0;
328 while (sent < resp.size()) {
329#if defined(_WIN32)
330 int n = ::send(client, resp.data() + sent, static_cast<int>(resp.size() - sent), 0);
331#else
332 int n = static_cast<int>(::send(client, resp.data() + sent, resp.size() - sent, 0));
333#endif
334 if (n <= 0) break;
335 sent += static_cast<std::size_t>(n);
336 }
337
338 close_socket(client);
339 }
340 }
